### 2.1 token_counter.py

```python
import tiktoken

_tokenizer = None

def _get_tokenizer(model: str = "qwen"):
    global _tokenizer
    if _tokenizer is None:
        # Qwen 使用 cl100k_base 近似（与 GPT-4 相同的 BPE tokenizer）
        _tokenizer = tiktoken.get_encoding("cl100k_base")
    return _tokenizer

def count_tokens(text: str, model: str = "qwen") -> int:
    """统计文本的 token 数"""
    if not text:
        return 0
    enc = _get_tokenizer(model)
    return len(enc.encode(text))
```

**注意**：Qwen 模型的精确 tokenizer 需要 `transformers` 的 QwenTokenizer，MVP 阶段用 tiktoken cl100k_base 近似（误差约 5-10%）。后续可替换为精确 tokenizer。

### 2.2 timestamp.py

```python
def format_timestamp(seconds: float, fmt: str = "mm:ss") -> str:
    """将秒数格式化为时间戳字符串"""
    if seconds < 0:
        seconds = 0
    hours = int(seconds // 3600)
    minutes = int((seconds % 3600) // 60)
    secs = int(seconds % 60)

    if fmt == "hh:mm:ss":
        return f"{hours:02d}:{minutes:02d}:{secs:02d}"
    else:  # mm:ss（默认）
        total_minutes = hours * 60 + minutes
        return f"{total_minutes:02d}:{secs:02d}"

def parse_timestamp(timestamp_str: str) -> float:
    """将时间戳字符串解析为秒数"""
    parts = timestamp_str.strip().split(":")
    if len(parts) == 2:  # mm:ss
        minutes, seconds = parts
        return int(minutes) * 60 + float(seconds)
    elif len(parts) == 3:  # hh:mm:ss
        hours, minutes, seconds = parts
        return int(hours) * 3600 + int(minutes) * 60 + float(seconds)
    else:
        raise InvalidTimestampError(f"无效时间戳格式: {timestamp_str}")
```

### 2.3 video_probe.py

```python
import subprocess
import json

def probe_video(video_path: str) -> VideoInfo:
    """探测视频文件信息"""
    cmd = [
        "ffprobe", "-v", "quiet",
        "-print_format", "json",
        "-show_format",
        "-show_streams",
        video_path
    ]
    result = subprocess.run(cmd, capture_output=True, text=True)
    if result.returncode != 0:
        raise FFprobeError(f"ffprobe 执行失败: {result.stderr}")

    data = json.loads(result.stdout)

    # 解析视频流
    video_stream = next((s for s in data["streams"] if s["codec_type"] == "video"), None)
    audio_stream = next((s for s in data["streams"] if s["codec_type"] == "audio"), None)

    return VideoInfo(
        path=video_path,
        duration=float(data["format"].get("duration", 0)),
        width=int(video_stream["width"]) if video_stream else 0,
        height=int(video_stream["height"]) if video_stream else 0,
        fps=eval(video_stream["r_frame_rate"]) if video_stream else 0,
        video_codec=video_stream["codec_name"] if video_stream else "",
        audio_codec=audio_stream["codec_name"] if audio_stream else "",
        audio_channels=int(audio_stream["channels"]) if audio_stream else 0,
        audio_sample_rate=int(audio_stream["sample_rate"]) if audio_stream else 0,
        size_bytes=int(data["format"].get("size", 0)),
        has_audio=audio_stream is not None,
        has_video=video_stream is not None,
    )
```

### 2.4 logger.py

```python
import logging
import os
from datetime import datetime

def setup_logger(name: str, log_dir: str = "./logs", level: str = "INFO",
                 console_output: bool = True) -> logging.Logger:
    """配置日志记录器"""
    logger = logging.getLogger(name)
    logger.setLevel(getattr(logging, level.upper()))

    # 避免重复添加 handler
    if logger.handlers:
        return logger

    formatter = logging.Formatter(
        "%(asctime)s [%(levelname)s] %(name)s: %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S"
    )

    # 文件输出
    os.makedirs(log_dir, exist_ok=True)
    log_file = os.path.join(log_dir, f"{name}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.log")
    file_handler = logging.FileHandler(log_file, encoding="utf-8")
    file_handler.setFormatter(formatter)
    logger.addHandler(file_handler)

    # 控制台输出
    if console_output:
        console_handler = logging.StreamHandler()
        console_handler.setFormatter(formatter)
        logger.addHandler(console_handler)

    return logger
```

### 2.5 file_utils.py

```python
import os
import hashlib

def ensure_dir(dir_path: str) -> None:
    """确保目录存在，不存在则创建"""
    if dir_path and not os.path.exists(dir_path):
        os.makedirs(dir_path, exist_ok=True)

def file_hash(file_path: str, algorithm: str = "md5") -> str:
    """计算文件的哈希值"""
    h = hashlib.new(algorithm)
    with open(file_path, "rb") as f:
        while chunk := f.read(8192):
            h.update(chunk)
    return h.hexdigest()
```
